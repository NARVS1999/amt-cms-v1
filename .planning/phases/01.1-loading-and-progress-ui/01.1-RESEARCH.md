### 2.2 Route Change Detection

Since this is a Next.js SSG app with `output: 'export'`, all routing is client-side. `usePathname()` from `next/navigation` is the detection mechanism.

**Pattern:**
```tsx
// RouteChangeLoader wraps admin layout children
'use client';
import { usePathname } from 'next/navigation';
import { useEffect, useState } from 'react';

export function RouteChangeLoader({ children }: { children: React.ReactNode }) {
  const pathname = usePathname();
  const [loading, setLoading] = useState(false);

  useEffect(() => {
    setLoading(true);
    // Use requestAnimationFrame to allow the browser to paint the overlay
    // before the next render cycle processes the new page
    const timeout = setTimeout(() => setLoading(false), 300);
    return () => clearTimeout(timeout);
  }, [pathname]);

  return (
    <div style={{ position: 'relative' }}>
      {loading && <div className="...overlay..." aria-hidden="true" />}
      {children}
    </div>
  );
}
```

**Key insight:** Since this is SSG, there's no router event like `routeChangeStart`/`routeChangeComplete`. The pathname change and React re-render are synchronous. To show the overlay, you must use either:
- `setTimeout(..., 0)` / `requestAnimationFrame` to defer the "hide" after the pathname change has rendered
- Or detect the pathname change in a ref and show overlay for a minimum duration

### 3.3 Page Transition Overlay (RouteChangeLoader)

```tsx
// components/admin/route-change-loader.tsx
'use client';

import { usePathname } from 'next/navigation';
import { useEffect, useRef, useState } from 'react';

export function RouteChangeLoader({ children }: { children: React.ReactNode }) {
  const pathname = usePathname();
  const prevPath = useRef(pathname);
  const [visible, setVisible] = useState(false);

  useEffect(() => {
    if (prevPath.current !== pathname) {
      prevPath.current = pathname;
      setVisible(true);
      const timer = setTimeout(() => setVisible(false), 400);
      return () => clearTimeout(timer);
    }
  }, [pathname]);

  return (
    <div style={{ position: 'relative', minHeight: 'inherit' }}>
      {visible && (
        <div
          className="absolute inset-0 z-[100] flex flex-col items-center justify-center gap-6 bg-white/90 transition-opacity duration-300"
          aria-hidden="true"
        >
          <div className="flex gap-2">
            {[0, 1, 2].map((i) => (
              <span
                key={i}
                className="h-3 w-3 rounded-full bg-[--color-primary]"
                style={{
                  animation: 'dotBounce 1.2s ease-in-out infinite',
                  animationDelay: `${i * 0.2}s`,
                }}
              />
            ))}
          </div>
          <span className="text-sm font-medium text-muted-foreground">Loading</span>
        </div>
      )}
      {children}
    </div>
  );
}
```

**Key CSS (in globals.css or via style tag):**
```css
@keyframes dotBounce {
  0%, 80%, 100% { transform: scale(0.6); opacity: 0.4; }
  40% { transform: scale(1); opacity: 1; }
}
```

### 3.4 Toast System — Provider + Hook

```tsx
// components/ui/toast.tsx
'use client';

import { createContext, useCallback, useContext, useState } from 'react';
import { cn } from '@/lib/utils';
import { X } from 'lucide-react';

type ToastVariant = 'success' | 'error' | 'info';

interface Toast {
  id: number;
  message: string;
  variant: ToastVariant;
}

interface ToastContextValue {
  showToast: (message: string, variant?: ToastVariant) => void;
}

const ToastContext = createContext<ToastContextValue | null>(null);

let toastId = 0;

export function ToastProvider({ children }: { children: React.ReactNode }) {
  const [toasts, setToasts] = useState<Toast[]>([]);

  const showToast = useCallback((message: string, variant: ToastVariant = 'info') => {
    const id = ++toastId;
    setToasts((prev) => [...prev, { id, message, variant }]);
    if (variant !== 'error') {
      setTimeout(() => {
        setToasts((prev) => prev.filter((t) => t.id !== id));
      }, 2000);
    }
  }, []);

  const dismiss = useCallback((id: number) => {
    setToasts((prev) => prev.filter((t) => t.id !== id));
  }, []);

  return (
    <ToastContext.Provider value={{ showToast }}>
      {children}
      <div
        className="fixed bottom-6 right-6 z-[9999] flex flex-col gap-2"
        aria-live="polite"
        aria-atomic="false"
      >
        {toasts.map((toast) => (
          <div
            key={toast.id}
            role="status"
            className={cn(
              'flex items-center gap-2 rounded-xl px-4 py-3 text-sm text-white shadow-lg',
              'animate-[toastSlideIn_0.25s_ease]',
              toast.variant === 'success' && 'bg-green-600',
              toast.variant === 'error' && 'bg-destructive',
              toast.variant === 'info' && 'bg-foreground',
            )}
            style={{ maxWidth: 320 }}
          >
            <span className="flex-1">{toast.message}</span>
            <button
              onClick={() => dismiss(toast.id)}
              className="ml-auto shrink-0 opacity-70 hover:opacity-100"
              aria-label="Dismiss"
            >
              <X size={14} aria-hidden="true" />
            </button>
          </div>
        ))}
      </div>
    </ToastContext.Provider>
  );
}

export function useToast() {
  const ctx = useContext(ToastContext);
  if (!ctx) throw new Error('useToast must be used within a ToastProvider');
  return ctx;
}
```

## 4. Edge Cases / Pitfalls

### 4.1 Route Change Timing (SSG-Specific)

Since `output: 'export'` means no server, all navigation is instant client-side re-renders. The pathname state change and React re-render happen synchronously in the same microtask. This means:
- Setting `loading=true` on pathname change AND `loading=false` with `setTimeout(0)` will never show the overlay because React batches state updates
- **Solution:** Use `requestAnimationFrame` + second state toggle, or a minimum 300ms display duration
- **Alternative:** Don't use an overlay at all — use the top bar (YouTube-style) approach which avoids the timing issue

### 4.2 NO `confirm()` — Accessibility

The `confirm()` browser dialog is:
- Not styleable (breaks brand consistency)
- Not accessible (no ARIA, keyboard trap issues)
- Blocks the main thread (cannot be async)
- **All pages using `confirm()` must be migrated to AlertDialog**

### 4.3 Error State Gaps

Several CRUD pages lack error handling entirely:
- Services: `catch` only handles UnauthorizedError, other errors are swallowed
- Team: Same pattern
- Pricing Plans: Same pattern
- Blog Posts: Has try/catch but uses `alert()` for non-auth errors
- Pages: Same as services

**Fix:** Add `error` state variable + error banner to all CRUD pages, consistent with dashboard/media pattern.

### 4.4 Button Loading State Pattern

Current pattern across pages:
```tsx
const [saving, setSaving] = useState(false);
```
```tsx
<Button onClick={handleSave} disabled={saving}>
  {saving ? 'Saving...' : 'Save'}
</Button>
```

This pattern is consistent and correct — just needs to be applied consistently:
- Login page: ✅ "Signing in..."
- Forgot password: ✅ "Sending..."
- Media upload: ✅ "Uploading..."
- Services: ❌ No loading state on Save button
- Team: ❌ No loading state on Save button
- Pricing Plans: ❌ No loading state on Save button
- Blog Posts: ✅ "Saving..." (via `saving` state)
- Pages: ❌ No loading state on Save button

### 4.5 Skeleton vs Inline Divs — Tradeoffs

**Using `<Skeleton className="h-4 w-full" />`** vs current `class="h-4 w-full animate-pulse rounded bg-muted"`:
- The component version is ~20 chars shorter, but more importantly it provides **one source of truth** for skeleton styling
- If animation timing or border radius changes, update one file
- Risk: If `Skeleton` is over-customized with `className`, the benefit diminishes

**Recommendation:** Use Skeleton component for all new/replaced skeleton instances. The component is cleaner for review + maintenance.

### 4.6 Progress Component — When to Use

The `progress.tsx` (shadcn) should be added to the registry but **only used** when:
- Upload progress tracking is added (future enhancement)
- Multi-step form progress (future)

Phase 01.1 scope does not include actual progress bar usage — just add the component.

### 4.7 Toast Positioning

The existing blog auto-save toast uses `fixed bottom-4 right-4`. The new toast system uses `fixed bottom-6 right-6` (slightly different). When migrating, the blog page's inline toast should be replaced with the shared `useToast()` hook.

**Edge case:** Blog posts page has TWO toast-like messages:
- `saveToast` ("Saved.") — success toast, auto-dismiss 2s
- `lastAutoSave` ("Draft saved") — also success toast, auto-dismiss 2s

Both should be migrated to call `showToast('Saved.', 'success')` and `showToast('Draft saved', 'success')`.

### 4.8 Accessibility Considerations

| Element | ARIA Requirement | Status |
|---------|-----------------|--------|
| Skeleton | `aria-hidden="true"` (decorative) — no role needed | ✅ Simple div, no ARIA needed |
| Spinner | `role="status"` + `sr-only` "Loading..." | ✅ Handled in component |
| Overlay | `aria-hidden="true"` — purely decorative | ✅ CSS class |
| Toast | `role="status"` + `aria-live="polite"` on container | ✅ In provider |
| Error banner | `role="alert"` + `aria-live="polite"` | ✅ Already in existing code |
| AlertDialog | Handled by base-ui (focus trap, ESC dismiss, aria-modal) | ✅ Already implemented |
| Input error | `aria-describedby` linking to error element | ⚠️ Login page already uses this; CRUD pages should adopt |
