<action>
1. **Create `components/LatestPosts.tsx`** as a server component:
   ```tsx
   import { fetchBlogPosts } from '@/lib/api';
   import { BlogCard } from './BlogCard';

   export async function LatestPosts() {
     const posts = await fetchBlogPosts();

     if (!posts || posts.length === 0) return null;

     const latest = posts.slice(0, 3);

     return (
       <section id="blog" className="py-20 bg-[var(--color-muted)]">
         <div className="container mx-auto px-4">
           <h2 className="text-3xl font-bold text-center text-[var(--color-foreground)]">Latest Insights</h2>
           <p className="mt-3 text-center text-[var(--color-muted-foreground)]">
             Tips, guides, and industry updates
           </p>
           <div className="mt-10 grid grid-cols-1 md:grid-cols-3 gap-8">
             {latest.map((post) => (
               <BlogCard
                 key={post.id}
                 title={post.title}
                 slug={post.slug}
                 excerpt={post.excerpt}
                 featured_image_url={post.featured_image_url}
                 published_at={post.published_at}
               />
             ))}
           </div>
           <div className="mt-8 text-center">
             <a
               href="/blog"
               className="text-[var(--color-primary)] hover:underline font-medium"
             >
               View All Posts →
             </a>
           </div>
         </div>
       </section>
     );
   }
   ```
2. The component returns `null` when 0 posts (follows `ServicesGrid` pattern).
</action>
